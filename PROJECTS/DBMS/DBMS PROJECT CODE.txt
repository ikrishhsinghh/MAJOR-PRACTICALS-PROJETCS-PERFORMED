
CREATE TABLE ServiceProvider1400 (
    Provider_ID NUMBER PRIMARY KEY,
    Provider_Name VARCHAR(15),
    Contact_Number VARCHAR(15)
);

CREATE TABLE Customer1400 (
    Customer_ID NUMBER PRIMARY KEY,
    Name VARCHAR(50),
    Mobile_Number VARCHAR(15),
    Email VARCHAR(50),
    City VARCHAR(30)
);


CREATE TABLE RechargePlan1400 (
    Plan_ID NUMBER PRIMARY KEY,
    Provider_ID NUMBER,
    Plan_Name VARCHAR(50),
    Validity VARCHAR(15),
    Price NUMBER(10,2),
    Data_Offer VARCHAR(30),
    FOREIGN KEY (Provider_ID) REFERENCES ServiceProvider1400(Provider_ID)
);


CREATE TABLE RechargeTransaction1400 (
    Transaction_ID NUMBER PRIMARY KEY,
    Customer_ID NUMBER,
    Plan_ID NUMBER,
    Recharge_Date DATE,
    Payment_Mode VARCHAR(15),
    Status VARCHAR(15),
    FOREIGN KEY (Customer_ID) REFERENCES Customer1400(Customer_ID),
    FOREIGN KEY (Plan_ID) REFERENCES RechargePlan1400(Plan_ID)
);

SELECT Plan_Name, Price, Validity, Data_Offer
FROM RechargePlan1400 RP
JOIN ServiceProvider1400 SP ON RP.Provider_ID = SP.Provider_ID
WHERE SP.Provider_Name = 'Jio';

SELECT T.Transaction_ID, P.Plan_Name, T.Recharge_Date, T.Payment_Mode
FROM RechargeTransaction1400 T
JOIN RechargePlan1400 P ON T.Plan_ID = P.Plan_ID
WHERE T.Customer_ID = 1 AND T.Status = 'Success';

SELECT T.Transaction_ID, C.Name AS Customer_Name, C.Mobile_Number, SP.Provider_Name, P.Plan_Name, P.Price, P.Validity, T.Recharge_Date, T.Payment_Mode, T.Status
FROM RechargeTransaction1400 T
JOIN Customer1400 C ON T.Customer_ID = C.Customer_ID
JOIN RechargePlan1400 P ON T.Plan_ID = P.Plan_ID
JOIN ServiceProvider1400 SP ON P.Provider_ID = SP.Provider_ID
ORDER BY T.Recharge_Date DESC;
